```jsx
import { useSelector, useDispatch } from "react-redux";
import FullCalendar from '@fullcalendar/react';
import dayGridPlugin from '@fullcalendar/daygrid';
import interactionPlugin from '@fullcalendar/interaction';
import { ChevronLeft, ChevronRight, Search } from "react-bootstrap-icons"; // Search 아이콘 추가
import { useState, useEffect, useRef } from "react";
import Sidebar from "../../components/include/Sidebar";
import Header from "../../components/include/Header";
import { removeEvent, loadEvents } from "../../redux/slice/calendarSlice";
import TopNav from "../include/TopNav";
import { useNavigate } from 'react-router-dom';
import { setCategory } from '../../redux/slice/calendarSlice';
import { categoryColors } from '../../utils/categoryColors';

const CalendarPage = () => {
  const navigate = useNavigate();  // react-router-dom을 통해 페이지 이동을 위해 navigate 훅 사용
  const dispatch = useDispatch();  // Redux 액션을 호출하기 위해 dispatch 훅 사용
  const events = useSelector((state) => state.calendar?.events || []);  // Redux에서 이벤트 목록 가져오기
  const category = useSelector((state) => state.calendar?.category);  // 현재 선택된 카테고리 가져오기
  const [currentMonth, setCurrentMonth] = useState(new Date().getMonth());  // 현재 월 상태 (초기값은 이번 달)
  const [selectedDate, setSelectedDate] = useState(null);  // 선택된 날짜 상태
  const [selectedEvents, setSelectedEvents] = useState([]);  // 선택된 날짜에 해당하는 이벤트들
  const [currentDate, setCurrentDate] = useState(`${new Date().getFullYear()}-${(new Date().getMonth() + 1).toString().padStart(2, '0')}-01`); // 달력 초기 날짜 (이번 달 1일)
  const [todayEvents, setTodayEvents] = useState([]);  // 오늘 날짜의 이벤트들
  const [searchQuery, setSearchQuery] = useState("");  // 검색어 상태
  const calendarRef = useRef(null);  // FullCalendar 참조 객체 (달력 API 사용을 위해)

  const today = new Date().toISOString().split('T')[0];  // 오늘 날짜 (ISO 형식)

  const handleSearchChange = (e) => {
    setSearchQuery(e.target.value);  // 검색어 상태를 업데이트하는 함수
  };

  const handleSearchKeyDown = (e) => {
    if (e.key === 'Enter') {
      // Enter 키를 눌렀을 때 검색어로 필터링된 일정 업데이트
      setSelectedEvents(events.filter(event => event.title.toLowerCase().includes(searchQuery.toLowerCase())));
  
      // 검색 후에 selectedDate가 설정되지 않으면 현재 날짜로 설정
      if (!selectedDate) {
        setSelectedDate(today); // 예시로 현재 날짜를 설정
      }
    }
  };
  

  useEffect(() => {
    const savedEvents = JSON.parse(localStorage.getItem('events')) || [];  // 로컬 스토리지에서 이전에 저장된 이벤트 로드
    dispatch(loadEvents(savedEvents));  // Redux store에 이벤트 로드
  }, [dispatch]);

  useEffect(() => {
    const filteredTodayEvents = events.filter(event => event.start?.split('T')[0] === today);  // 오늘 날짜에 해당하는 이벤트 필터링
    setTodayEvents(filteredTodayEvents);  // 오늘의 이벤트 상태 업데이트
  }, [events]);

  useEffect(() => {
      const filteredEvents = events.filter(event => {
      const isSameDate = event.start?.split('T')[0] === selectedDate;  // 선택된 날짜에 해당하는 이벤트 필터링
      const matchesSearchQuery = event.title.toLowerCase().includes(searchQuery.toLowerCase());  // 검색어와 일치하는 이벤트 필터링
      return isSameDate && matchesSearchQuery;
    });
    setSelectedEvents(filteredEvents);  // 필터링된 이벤트들 상태 업데이트
  }, [events, selectedDate, searchQuery]);
  

  useEffect(() => {
    const firstDateOfMonth = new Date(new Date().getFullYear(), currentMonth, 1);  // 현재 월의 첫 날짜
    const lastDateOfMonth = new Date(new Date().getFullYear(), currentMonth + 1, 0);  // 현재 월의 마지막 날짜
    const filteredEvents = events.filter(event => {
      const eventDate = new Date(event.start);  // 이벤트 날짜
      return eventDate >= firstDateOfMonth && eventDate <= lastDateOfMonth;  // 현재 월 내의 이벤트만 필터링
    });
    setSelectedEvents(filteredEvents);  // 현재 월에 해당하는 이벤트들 상태 업데이트
  }, [currentMonth, events]);

  useEffect(() => {
    setSelectedDate(`${new Date().getFullYear()}-${(currentMonth + 1).toString().padStart(2, '0')}-01`);  // 월 변경 시 첫날로 선택된 날짜 설정
  }, [currentMonth]);

  useEffect(() => {
    dispatch(setCategory('all'));  // 카테고리가 변경될 때마다 카테고리를 'all'로 설정
  }, [dispatch]);

  const filteredEvents = events
    .filter(event => category === "all" || event.category === category)  // 카테고리에 따라 이벤트 필터링
    .map(event => ({
      ...event,
      backgroundColor: categoryColors[event.category] || '#D1D5DB',  // 이벤트의 배경 색상 설정
      borderColor: 'none',
      borderWidth: 0,
      borderStyle: 'none',
    }));

  const handlePrevMonth = () => {
    if (currentMonth > 0) {
      const newMonth = currentMonth - 1;  // 이전 달로 이동
      setCurrentMonth(newMonth);
      moveCalendarToMonth(newMonth);
    }
  };

  const handleNextMonth = () => {
    if (currentMonth < 11) {
      const newMonth = currentMonth + 1;  // 다음 달로 이동
      setCurrentMonth(newMonth);
      moveCalendarToMonth(newMonth);
    }
  };

  const moveCalendarToMonth = (month) => {
    const year = new Date().getFullYear();  // 현재 년도
    const newDate = new Date(year, month, 1);  // 새로운 월의 첫날
    if (calendarRef.current) {
      calendarRef.current.getApi().gotoDate(newDate);  // FullCalendar의 API를 사용하여 해당 날짜로 이동
    }
  };

  const handleDateClick = (info) => {
    const clickedDate = info.dateStr;  // 클릭한 날짜
    const calendarApi = calendarRef.current.getApi();
    if (calendarApi.getDate().toISOString().split('T')[0] === today) {
      calendarApi.unselect();  // 오늘 날짜 클릭 시 선택 해제
    }
    if (clickedDate !== today) {
      calendarApi.select(clickedDate);  // 오늘이 아닌 날짜 클릭 시 선택
    } else {
      calendarApi.unselect();  // 오늘 날짜 다시 클릭하면 선택 해제
    }
    setSelectedDate(clickedDate);  // 선택된 날짜 상태 업데이트
    setSelectedEvents(events.filter(event => event.start?.split('T')[0] === clickedDate));  // 해당 날짜의 이벤트 상태 업데이트
  };

  const handleEventClick = (info) => {
    const eventId = info.event.id;  // 클릭한 이벤트의 ID
    if (eventId) {
      navigate(`/event/${eventId}`);  // 이벤트 상세 페이지로 이동
    } else {
      alert("잘못된 이벤트입니다.");  // 이벤트 ID가 없으면 경고 메시지 표시
    }
  };

  const handleTodayClick = () => {
    const calendarApi = calendarRef.current.getApi();
    calendarApi.gotoDate(today);  // 오늘 날짜로 이동

    const todayDate = new Date(today);
    const todayMonth = todayDate.getMonth();
    setCurrentMonth(todayMonth);  // 현재 월을 오늘로 설정

    setSelectedDate(today);  // 선택된 날짜를 오늘로 설정
  };

  return (
    <div className="flex h-screen bg-gray-100">
      <Sidebar />  {/* 사이드바 컴포넌트 */}
      <div className="flex-1 p-6">
        <Header />  {/* 헤더 컴포넌트 */}
        <TopNav />  {/* 상단 네비게이션 바 컴포넌트 */}
        <div className="flex">
          <div className="flex-1 bg-white p-6 rounded-2xl shadow-lg">
            <div className="flex justify-between items-center mb-4">
              {/* 왼쪽: 오늘 버튼, 이전/다음 월 버튼 */}
              <div className="flex items-center space-x-2">
                <button onClick={handleTodayClick} className="px-4 py-2 bg-[#006D2C] text-white rounded-md">오늘</button>
                <button onClick={handlePrevMonth} disabled={currentMonth === 0}>
                  <ChevronLeft size={24} />
                </button>
                <h2 className="text-lg font-bold">{currentMonth + 1}월</h2>
                <button onClick={handleNextMonth} disabled={currentMonth === 11}>
                  <ChevronRight size={24} />
                </button>
              </div>

              {/* 오른쪽: 검색창 및 검색 아이콘 */}
              <div className="flex items-center space-x-2">
                <input
                  type="text"
                  placeholder="이벤트 검색"
                  className="w-64 p-2 border rounded-md"
                  onChange={handleSearchChange}
                  onKeyDown={handleSearchKeyDown} // Enter 키 이벤트 추가
                />
                <Search size={24} className="cursor-pointer" onClick={() => handleSearchKeyDown({ key: 'Enter' })} /> {/* 검색 아이콘 */}
              </div>
            </div>

            <FullCalendar
              ref={calendarRef}
              plugins={[dayGridPlugin, interactionPlugin]}
              initialView="dayGridMonth"
              events={filteredEvents}
              headerToolbar={false}
              height="auto"
              initialDate={currentDate}
              dateClick={handleDateClick}
              eventClick={handleEventClick}
            />
          </div>

          <div className="flex flex-col space-y-4 w-96 ml-6 p-4 bg-white rounded-2xl shadow-lg">
            {selectedDate ? (
              <div>
                <h3 className="text-lg font-bold">{selectedDate} 일정</h3>
                {selectedEvents.length > 0 ? (
                  <ul>
                    {selectedEvents.map((event) => (
                      <li key={event.id} className="mt-2 p-2 bg-gray-100 rounded-md">
                        {event.title}
                      </li>
                    ))}
                  </ul>
                ) : (
                  <p>등록된 일정이 없습니다.</p>
                )}
              </div>
            ) : (
              <div>
                <h3 className="text-lg font-bold">오늘의 일정</h3>
                {todayEvents.length > 0 ? (
                  <ul>
                    {todayEvents.map((event) => (
                      <li key={event.id} className="mt-2 p-2 bg-gray-100 rounded-md">
                        {event.title}
                      </li>
                    ))}
                  </ul>
                ) : (
                  <p>오늘의 일정이 없습니다.</p>
                )}
              </div>
            )}
            <button
              className="w-full p-2 mt-4 bg-[#006D2C] text-white rounded-md"
              onClick={() => navigate(`/write?date=${selectedDate || today}`)}  // 일정 추가 페이지로 이동
            >
              일정 추가
            </button>
          </div>
        </div>
      </div>
    </div>
  );
};

export default CalendarPage;

```
